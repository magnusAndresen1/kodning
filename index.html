import time #Import af biblioteker
import machine
from machine import Pin

trig_pin = Pin(7, Pin.OUT) #Initialisering af pins
echo_pin = Pin(6, Pin.IN)
LED = Pin(5, Pin.OUT)

def afstand():#til at måle afstand
    trig_pin.value(0)#Sluk triggeren
    time.sleep_us(2)
    trig_pin.value(1)#Send ultralydssignalet
    time.sleep_us(10)
    trig_pin.value(0)
    
    duration = machine.time_pulse_us(echo_pin, 1, 30000)#Mål tiden
    
    return duration# start tid forfra

while True:
    tid = afstand()#Start målingen
    print(tid) #udskriv tid
    
    meter = round(342.00 * tid / 2000000, 2)#beregn afstand
    print(meter, "m") #printer afstand
    
    if meter < 1: #Tænd eller sluk LED'en
        LED.value(1)
    else:
        LED.value(0)
        
    time.sleep(2) #sluk

Refleksion & Læring

  • Det var en udfordrende kode, jeg kunne lidt af starten, men der efter måtte jeg søge hjælp hos Jørgen og en ai til at give hints til hvad jeg skulle gøre.

// Eksempel på koden der blev skrevet
int main() {
    from machine import Pin
import time
# Step 1: Initialiser knap og LED
knap = Pin(7, Pin.IN, Pin.PULL_DOWN)   # Knap på pin 7
led = Pin(5, Pin.OUT)                  # LED på pin 5

while True:
    # Step 2: Vent på at knappen bliver trykket ned
    if knap.value() == 1:
        # Step 3: Når knappen er trykket ned, tænd LED'en
        led.value(1)
        print("Knap trykket - LED tændt")

        # Step 4: Vent et stykke tid
        time.sleep(0.5)

        # Step 5: Sluk LED'en
        led.value(0)
        print("LED slukket")

        # Step 6: Gentag fra step 2
        time.sleep(0.2)  # lille pause for at undgå dobbelttryk

}

// Eksempel på koden der blev skrevet
import time
from machine import Pin   # Biblioteker

trig = Pin(7, Pin.OUT)    # Trigger-pin
echo = Pin(6, Pin.IN)     # Echo-pin
led  = Pin(5, Pin.OUT)    # LED-advarsel

def afstand():
    trig.value(0)         # Start lav
    time.sleep_us(2)
    trig.value(1)         # Send puls
    time.sleep_us(10)
    trig.value(0)

    while echo.value() == 0:
        start = time.ticks_us()   # Puls starter
    while echo.value() == 1:
        slut = time.ticks_us()    # Puls slutter

    return (slut - start) * 34300 / 2_000_000   # Beregn afstand

while True:
    a = afstand()
    print(a, "cm")        # Vis afstand

    led.value(1 if a < 30 else 0)  # LED ved tæt afstand
    time.sleep(0.2)       # Lille pause
}

Refleksion & Læring

Kildekode / Resultat

// Eksempel på koden der blev skrevet
int main() {
    from machine import Pin
import neopixel
from time import sleep

button = Pin(11, Pin.IN, Pin.PULL_UP)

# Timing til WS2812B-MINI-V6
timing_v6 = (300, 950, 700, 600)

antal_led = 5

leds = neopixel.NeoPixel(
    Pin(10, Pin.OUT),
    antal_led,
    bpp=3,
    timing=timing_v6
)

while True:
    for i in range(antal_led):
        leds[i] = (255, 0, 0)
        leds.write()
        sleep(0.5)

    for i in range(antal_led):
        leds[i] = (0, 255, 0)
        leds.write()
        sleep(0.5)

    for i in range(antal_led):
        leds[i] = (0, 0, 255)
        leds.write()
        sleep(0.5)
}
